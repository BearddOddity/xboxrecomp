cmake_minimum_required(VERSION 3.20)
project(nv2a_pgraph_trap_regression C)
enable_testing()
if(NOT WIN32)
    message(STATUS "nv2a_pgraph_trap: Windows only (vectored exceptions); skipped")
    return()
endif()
add_executable(nv2a-pgraph-trap-test nv2a_pgraph_trap_test.c
    ../../src/kernel/nv2a_pgraph_trap.c)
target_include_directories(nv2a-pgraph-trap-test PRIVATE ../../src/kernel ../../src)
if(MSVC)
    target_compile_definitions(nv2a-pgraph-trap-test PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_test(NAME nv2a-pgraph-trap COMMAND nv2a-pgraph-trap-test)
